45uint8_t OS_TaskCreate(OS_TaskFunction_t pTaskCode, uint8_t u8Prio, uint8_t u8TaskId){
46 if(g_u8TaskCount >= Max_Task ) return 1; // Buffer ist voll
47
48 TCB_sctTCB_t *pNewTask = &g_asctTCBs[g_u8TaskCount];
49 pNewTask->u8TaskId = u8TaskId;
50 pNewTask->u8TaskPrio = u8Prio;
51 pNewTask->eTaskState = TaskState_Blocked;
52 pNewTask->u32DelayTicks = 0;
53 pNewTask->u32TimeoutTicks = 0;
54
55
56 uint32_t *pStack = &pNewTask->au32TaskStack[TCB_TASK_STACK_SIZE];
57
58 // 1. Hardware Stack Frame (Wird beim bx lr automatisch von der CPU geladen)
59 pStack--; *pStack = 0x01000000; // xPSR (Thumb-Bit MUSS 1 sein)
60 pStack--; *pStack = (uint32_t)pTaskCode; // PC (Startadresse des Tasks)
61 pStack--; *pStack = 0xFFFFFFF9; // LR (EXC_RETURN: Zurück in Thread Mode mit PSP)
62 pStack--; *pStack = 0x00000000; // R12
63 pStack--; *pStack = 0x00000000; // R3
64 pStack--; *pStack = 0x00000000; // R2
65 pStack--; *pStack = 0x00000000; // R1
66 pStack--; *pStack = 0x00000000; // R0
67
68 // 2. Software Stack Frame
69 pStack--; *pStack = 0x00000000; // R11
70 pStack--; *pStack = 0x00000000; // R10
71 pStack--; *pStack = 0x00000000; // R9
72 pStack--; *pStack = 0x00000000; // R8
73 pStack--; *pStack = 0x00000000; // R7
74 pStack--; *pStack = 0x00000000; // R6
75 pStack--; *pStack = 0x00000000; // R5
76 pStack--; *pStack = 0x00000000; // R4
77
78 // Aktuellen Stack-Pointer im TCB speichern
79 pNewTask->u32TaskSP = (uint32_t)pStack;
80
81 g_u8TaskCount++;
82
83 SEGGER_SYSVIEW_OnTaskCreate((uint32_t)pNewTask); //Task in der internen SystemView-Datenbank registrieren
84
85 return 0; //Erfolg
86}

96void OS_KernelStart(void){
97 if(g_u8TaskCount == 0) return;
98
99 // Task mit der höchsten Prio suchen
100 TCB_sctTCB_t *pPrioTask = &g_asctTCBs[0];
101
102 for (uint8_t i = 1; i < g_u8TaskCount; i++) {
103 // Kleinere Zahl = Höhere Priorität
104 if (g_asctTCBs[i].u8TaskPrio < pPrioTask->u8TaskPrio) {
105 pPrioTask = &g_asctTCBs[i];
106 }
107 }
108
109 g_pCurrentTCB = pPrioTask;
110 g_pCurrentTCB->eTaskState = TaskState_Running;
111 SEGGER_SYSVIEW_OnTaskStartExec((uint32_t)g_pCurrentTCB); // Sagt SystemView: Die neue Task läuft jetzt

192__attribute__((naked)) void OS_StartFirstTask(void)
193{
194 __asm volatile (
195
196 "ldr r0, =g_pCurrentTCB \n" // Lade Adresse des TCB-Pointers
197 "ldr r1, [r0] \n" // Dereferenziere: r1 = Adresse des aktuellen TCBs
198 "ldr r2, [r1] \n" // Lade erstes Element des TCBs (u32TaskSP) in r2
199
200 "msr msp, r2 \n" // Schreibe r2 in den Process Stack Pointer (PSP)
201
202 "pop {r4-r11} \n"
203 "pop {r0-r3, r12, lr} \n"
204 "pop {r3} \n"
205 "pop {r2} \n"
206
207 "msr apsr_nzcvq, r2 \n" // Hier werden die Flags wieder hergestellt
208 "bx r3 \n"
209
210 );
211}
