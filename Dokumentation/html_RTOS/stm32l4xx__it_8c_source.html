227void SysTick_Handler(void)
228{
229 /* USER CODE BEGIN SysTick_IRQn 0 */
230 SEGGER_SYSVIEW_RecordEnterISR(); // gib die exakte ID = 15 vom SysTick-Interrupt, Ziel: überprüfen, wie lange der CPU im ISR verbringt
231 /* USER CODE END SysTick_IRQn 0 */
232 HAL_IncTick();
233 /* USER CODE BEGIN SysTick_IRQn 1 */
234 // Wird überprüft, ob der Timeout abgelaufen ist
235 for(uint8_t i = 0; i < Max_Tasks; i++) {
236 if (g_asctTCBs[i].eTaskState == TaskState_Blocked && g_asctTCBs[i].u32TimeoutTicks > 0) {
237 g_asctTCBs[i].u32TimeoutTicks--; // Timer herunterzählen
238
239 if (g_asctTCBs[i].u32TimeoutTicks == 0) {
240 // Timeout ist abgelaufen! Task aufwecken.
241 g_asctTCBs[i].eTaskState = TaskState_Ready;
242 SEGGER_SYSVIEW_OnTaskStartReady((uint32_t)&g_asctTCBs[i]);
243 }
244 }
245 }
246
247
248 // Hier wird überprüft, ob die Schalfzeit von den blockierende Task abgelaufen ist, wenn ja, wird er aufgewacht.
249 for(int i = 0;i < Max_Tasks;i++){
250 if(g_asctTCBs[i].eTaskState == TaskState_Blocked)
251 {
252
253
254 if(g_asctTCBs[i].u32DelayTicks == 0){
255 g_asctTCBs[i].eTaskState = TaskState_Ready;
256 SEGGER_SYSVIEW_OnTaskStartReady((uint32_t)&g_asctTCBs[i]);
257 break;
258 }
259 g_asctTCBs[i].u32DelayTicks--;
260 }
261
262 }
263 // Hier wird überprüft, ob die Zeit der Präemtive RR abgelaufen ist, wenn ja wird der PendSV-Interrupt gesetzt
264 if(g_u8PraemtiveTime > 0) g_u8PraemtiveTime--;
265 else if (g_u8PraemtiveTime == 0) {
266 g_u8PraemtiveTime = TIME_SLICE_TICKS;
267 //g_asctTCBs[0].eTaskState = TaskState_Blocked;
268 //g_asctTCBs[1].eTaskState = TaskState_Ready;
269 // Löst den PendSV-Interrupt aus, um den Task zu wechseln
270 SCB->ICSR |= SCB_ICSR_PENDSVSET_Msk;
271 }
272
273 SEGGER_SYSVIEW_RecordExitISR(); // Hier wird SystemView mitgeteilt, dass die ISR in diesem Moment verlassen wurde
274 /* USER CODE END SysTick_IRQn 1 */
275}
