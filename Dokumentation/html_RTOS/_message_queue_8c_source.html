80uint8_t OS_MsgQ_SendNonBlocking(OS_MessageQueue_t* pQ, uint32_t u32Msg) {
81 __disable_irq();
82
83 // SYSTEMVIEW : Send-Event protokollieren
84 SEGGER_SYSVIEW_RecordU32(SYSVIEW_EV_QUEUE_SEND, (uint32_t)pQ); // Damit weiß ich genaue in welcher MessageQueue, der Task schreibt
85
86
87 if (OS_MsgQ_IsFull(pQ)) {
88 __enable_irq();
89 return 0; // Queue voll
90 }
91
92 pQ->pBuffer[pQ->u32Head] = u32Msg;
93 pQ->u32Head = (pQ->u32Head + 1) % pQ->u32Capacity;
94 pQ->u32Count++;
95
96 // Wartenden Leser aufwecken, falls vorhanden
97 if (pQ->u32WaitingReaders != 0) {
98 for (uint8_t i = 0; i < Max_Task; i++) { // Prioritätsbasierter Zugriff
99 if (pQ->u32WaitingReaders & (1 << i)) {
100 pQ->u32WaitingReaders &= ~(1 << i);
101 g_asctTCBs[i].eTaskState = TaskState_Ready;
102 SEGGER_SYSVIEW_OnTaskStartReady((uint32_t)&g_asctTCBs[i]);
103 if (g_pCurrentTCB->eTaskState == TaskState_Running) {
104 g_pCurrentTCB->eTaskState = TaskState_Ready;
105 SEGGER_SYSVIEW_OnTaskStartReady((uint32_t)g_pCurrentTCB);
106 }
107 __enable_irq();
108 SCB->ICSR |= SCB_ICSR_PENDSVSET_Msk; // Context Switch anfordern
109 return 1;
110 }
111 }
112 }
113
114 __enable_irq();
115 return 1;
116}
